= Biblioteca ROOM

O SDK do Android fornece uma biblioteca que facilita o uso do banco de dados. Conforme consta na documentação oficial, a biblioteca ROOM adiciona uma camada de abstração retirando as implementações mais complexas do banco SQLite. Para mais detalhes, acesse a página: GOOGLE DEVELOPERS. Salvar dados em um banco de dados local usando Room. Disponível em: https://developer.android.com/training/data-storage/room?hl=pt-br. Acessado em 20 ago. 2025.

Para usar a lib ROOM no seu projeto, primeiramente é necessário importar a biblioteca. Dessa forma, deve-se ir no arquivo Gradle a nível de app e adicionar as duas linhas no bloco de dependências:

[source,xml]
implementation ("androidx.room:room-runtime:2.4.0")
annotationProcessor ("androidx.room:room-compiler:2.4.0")

ou

[source,xml]
implementation("androidx.room:room-runtime:2.7.0-rc03")
annotationProcessor("androidx.room:room-compiler:2.7.0-rc03")

import android.os.Parcel;
import android.os.Parcelable;
import androidx.room.ColumnInfo;
import androidx.room.Entity;
import androidx.room.PrimaryKey;
@Entity
public class Pessoa implements Parcelable {
    @PrimaryKey(autoGenerate = true)
    private int id;
    @ColumnInfo(name = "nome")
    private String nome;
    @ColumnInfo(name = "telefone")
    private String telefone;
    public Pessoa(int id, String nome, String telefone) {
        this.id = id;
        this.nome = nome;
        this.telefone = telefone;
    }
    public Pessoa() {}
    protected Pessoa(Parcel in) {
        id = in.readInt();
        nome = in.readString();
        telefone = in.readString();
    }
    public static final Creator<Pessoa> CREATOR = new Creator<Pessoa>() {
        @Override
        public Pessoa createFromParcel(Parcel in) {
            return new Pessoa(in);
        }
        @Override
        public Pessoa[] newArray(int size) {
            return new Pessoa[size];
        }
    };
    public int getId() { return id; }
    public void setId(int id) { this.id = id; }
    public String getNome() { return nome; }
    public void setNome(String nome) { this.nome = nome; }
    public String getTelefone() { return telefone; }
    public void setTelefone(String telefone) { this.telefone = telefone; }
    @Override
    public String toString() {
        return nome + " - " + telefone;
    }
    @Override
    public int describeContents() {
        return 0;
    }
    @Override
    public void writeToParcel(Parcel parcel, int flags) {
        parcel.writeInt(id);
        parcel.writeString(nome);
        parcel.writeString(telefone);
    }
}

import android.content.Context;
import androidx.room.Room;
public class BancoProvider {
    private static volatile Banco INSTANCE;
    public static Banco getDatabase(Context context) {
        if (INSTANCE == null) {
            synchronized (BancoProvider.class) {
                if (INSTANCE == null) {
                    INSTANCE = Room.databaseBuilder(
                            context.getApplicationContext(),
                            Banco.class,
                            "meu_banco"
                    ).build();
                }
            }
        }
        return INSTANCE;
    }
}

import android.content.Intent;
import android.os.Bundle;
import android.os.Handler;
import android.os.Looper;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;
import androidx.appcompat.app.AppCompatActivity;
import java.util.ArrayList;
import java.util.List;
public class MainActivity extends AppCompatActivity {
    private EditText editTextNome, editTextTelefone;
    private Button buttonS, buttonL, buttonD;
    private Pessoa p;
    private PessoaDao dao;
    private Handler mainHandler;
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);
        mainHandler = new Handler(Looper.getMainLooper());
        editTextNome = findViewById(R.id.editTextNome);
        editTextTelefone = findViewById(R.id.editTextTelefone);
        buttonS = findViewById(R.id.buttonS);
        buttonD = findViewById(R.id.buttonD);
        buttonL = findViewById(R.id.buttonL);
        if (android.os.Build.VERSION.SDK_INT >= android.os.Build.VERSION_CODES.TIRAMISU) {
            p = getIntent().getParcelableExtra("dado", Pessoa.class);
        } else {
            p = getIntent().getParcelableExtra("dado");
        }
        if (p != null) {
            editTextNome.setText(p.getNome());
            editTextTelefone.setText(p.getTelefone());
            buttonD.setEnabled(true);
        } else {
            buttonD.setEnabled(false);
        }
        dao = BancoProvider.getDatabase(this).pessoaDao();
    }
    public void clicar(View view) {
        int id = view.getId();
        if (id == R.id.buttonS) {
            String nome = editTextNome.getText().toString().trim();
            String telefone = editTextTelefone.getText().toString().trim();
            new Thread(new Runnable() {
                @Override
                public void run() {
                    try {
                        if (p == null) {
                            Pessoa nova = new Pessoa();
                            nova.setNome(nome);
                            nova.setTelefone(telefone);
                            final long resultado = dao.adicionar(nova);
                            mainHandler.post(new Runnable() {
                                @Override
                                public void run() {
                                    if (resultado != -1) {
                                        Toast.makeText(MainActivity.this, "Salvo com sucesso", Toast.LENGTH_SHORT).show();
                                        limparCampos();
                                    } else {
                                        Toast.makeText(MainActivity.this, "Erro ao salvar", Toast.LENGTH_SHORT).show();
                                    }
                                }
                            });
                        } else {
                            p.setNome(nome);
                            p.setTelefone(telefone);
                            int resultado = dao.atualizar(p);
                            mainHandler.post(new Runnable() {
                                @Override
                                public void run() {
                                    if (resultado > 0) {
                                        Toast.makeText(MainActivity.this, "Atualizado com sucesso", Toast.LENGTH_SHORT).show();
                                        limparCampos();
                                        p = null;
                                        buttonD.setEnabled(false);
                                    } else {
                                        Toast.makeText(MainActivity.this, "Erro ao atualizar", Toast.LENGTH_SHORT).show();
                                    }
                                }
                            });
                        }
                    } catch (Exception e) {
                        mainHandler.post(new Runnable() {
                            @Override
                            public void run() {
                                Toast.makeText(MainActivity.this, "Erro: " + e.getMessage(), Toast.LENGTH_SHORT).show();
                            }
                        });
                    }
                }
            }).start();
        } else if (id == R.id.buttonL) {
            new Thread(new Runnable() {
                @Override
                public void run() {
                    try {
                        List<Pessoa> lista = dao.listar();
                        ArrayList<Pessoa> dados = new ArrayList<>(lista);
                        mainHandler.post(new Runnable() {
                            @Override
                            public void run() {
                                if (!dados.isEmpty()) {
                                    Intent intent = new Intent(MainActivity.this, SegundaActivity.class);
                                    intent.putParcelableArrayListExtra("dados", dados);
                                    startActivity(intent);
                                    finish();
                                } else {
                                    Toast.makeText(MainActivity.this, "Nenhum registro cadastrado", Toast.LENGTH_SHORT).show();
                                }
                            }
                        });
                    } catch (Exception e) {
                        mainHandler.post(new Runnable() {
                            @Override
                            public void run() {
                                Toast.makeText(MainActivity.this, "Erro ao listar: " + e.getMessage(), Toast.LENGTH_SHORT).show();
                            }
                        });
                    }
                }
            }).start();
        } else if (id == R.id.buttonD) {
            if (p == null) {
                Toast.makeText(this, "Selecione um registro para excluir", Toast.LENGTH_SHORT).show();
                return;
            }
            new Thread(new Runnable() {
                @Override
                public void run() {
                    try {
                        int resultado = dao.excluir(p);
                        mainHandler.post(new Runnable() {
                            @Override
                            public void run() {
                                if (resultado > 0) {
                                    Toast.makeText(MainActivity.this, "Excluído com sucesso", Toast.LENGTH_SHORT).show();
                                    limparCampos();
                                    p = null;
                                    buttonD.setEnabled(false);
                                } else {
                                    Toast.makeText(MainActivity.this, "Erro ao excluir", Toast.LENGTH_SHORT).show();
                                }
                            }
                        });
                    } catch (Exception e) {
                        mainHandler.post(new Runnable() {
                            @Override
                            public void run() {
                                Toast.makeText(MainActivity.this, "Erro: " + e.getMessage(), Toast.LENGTH_SHORT).show();
                            }
                        });
                    }
                }
            }).start();
        }
    }
    private void limparCampos() {
        editTextNome.setText("");
        editTextTelefone.setText("");
    }
}

- MainActivity.kt
[source,kotlin]
import android.content.Intent
import android.os.Build
import android.os.Bundle
import android.view.View
import android.widget.Button
import android.widget.EditText
import android.widget.Toast
import androidx.appcompat.app.AppCompatActivity
import androidx.lifecycle.lifecycleScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.launch
import kotlinx.coroutines.withContext
class MainActivity : AppCompatActivity() {
    private lateinit var editTextNome: EditText
    private lateinit var editTextTelefone: EditText
    private lateinit var buttonS: Button
    private lateinit var buttonL: Button
    private lateinit var buttonD: Button
    private var p: Pessoa? = null
    private lateinit var dao: PessoaDao
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        editTextNome = findViewById(R.id.editTextNome)
        editTextTelefone = findViewById(R.id.editTextTelefone)
        buttonS = findViewById(R.id.buttonS)
        buttonD = findViewById(R.id.buttonD)
        buttonL = findViewById(R.id.buttonL)
        p = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU) {
            intent.getParcelableExtra("dado", Pessoa::class.java)
        } else {
            @Suppress("DEPRECATION")
            intent.getParcelableExtra("dado")
        }
        p?.let {
            editTextNome.setText(it.nome)
            editTextTelefone.setText(it.telefone)
            buttonD.isEnabled = true
        } ?: run {
            buttonD.isEnabled = false
        }
        dao = BancoProvider.getDatabase(this).pessoaDao()
    }
    fun clicar(view: View) {
        when (view.id) {
            R.id.buttonS -> salvarOuAtualizar()
            R.id.buttonL -> listarPessoas()
            R.id.buttonD -> excluirPessoa()
        }
    }
    private fun salvarOuAtualizar() {
        val nome = editTextNome.text.toString().trim()
        val telefone = editTextTelefone.text.toString().trim()
        // Executa a tarefa de banco de dados em uma coroutine gerenciada pelo ciclo de vida
        lifecycleScope.launch {
            try {
                if (p == null) {
                    val nova = Pessoa(nome = nome, telefone = telefone)
                    val resultado = withContext(Dispatchers.IO) { dao.adicionar(nova) }
                    if (resultado != -1L) {
                        Toast.makeText(this@MainActivity, "Salvo com sucesso", Toast.LENGTH_SHORT).show()
                        limparCampos()
                    } else {
                        Toast.makeText(this@MainActivity, "Erro ao salvar", Toast.LENGTH_SHORT).show()
                    }
                } else {
                    p?.apply {
                        this.nome = nome
                        this.telefone = telefone
                    }
                    val resultado = withContext(Dispatchers.IO) { dao.atualizar(p!!) }
                    if (resultado > 0) {
                        Toast.makeText(this@MainActivity, "Atualizado com sucesso", Toast.LENGTH_SHORT).show()
                        limparCampos()
                        p = null
                        buttonD.isEnabled = false
                    } else {
                        Toast.makeText(this@MainActivity, "Erro ao atualizar", Toast.LENGTH_SHORT).show()
                    }
                }
            } catch (e: Exception) {
                Toast.makeText(this@MainActivity, "Erro: ${e.message}", Toast.LENGTH_SHORT).show()
            }
        }
    }
    private fun listarPessoas() {
        lifecycleScope.launch {
            try {
                val lista = withContext(Dispatchers.IO) { dao.listar() }
                val dados = ArrayList(lista)
                if (dados.isNotEmpty()) {
                    val intent = Intent(this@MainActivity, SegundaActivity::class.java).apply {
                        putParcelableArrayListExtra("dados", dados)
                    }
                    startActivity(intent)
                    finish()
                } else {
                    Toast.makeText(this@MainActivity, "Nenhum registro cadastrado", Toast.LENGTH_SHORT).show()
                }
            } catch (e: Exception) {
                Toast.makeText(this@MainActivity, "Erro ao listar: ${e.message}", Toast.LENGTH_SHORT).show()
            }
        }
    }
    private fun excluirPessoa() {
        val pessoaParaExcluir = p ?: run {
            Toast.makeText(this, "Selecione um registro para excluir", Toast.LENGTH_SHORT).show()
            return
        }
        lifecycleScope.launch {
            try {
                val resultado = withContext(Dispatchers.IO) { dao.excluir(pessoaParaExcluir) }
                if (resultado > 0) {
                    Toast.makeText(this@MainActivity, "Excluído com sucesso", Toast.LENGTH_SHORT).show()
                    limparCampos()
                    p = null
                    buttonD.isEnabled = false
                } else {
                    Toast.makeText(this@MainActivity, "Erro ao excluir", Toast.LENGTH_SHORT).show()
                }
            } catch (e: Exception) {
                Toast.makeText(this@MainActivity, "Erro: ${e.message}", Toast.LENGTH_SHORT).show()
            }
        }
    }
    private fun limparCampos() {
        editTextNome.setText("")
        editTextTelefone.setText("")
    }
}

import android.os.Bundle
import android.util.Log
import androidx.appcompat.app.AppCompatActivity
import androidx.room.Room
class MainActivity : AppCompatActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)
        val db = Room.databaseBuilder(
            applicationContext,
            AppDatabase::class.java,
            "escola_n_n.db"
        ).allowMainThreadQueries().build()
        val dao = db.appDao()
        // 1. Inserir Curso e Disciplinas
        val cursoId = dao.inserirCurso(Curso("Engenharia de Software"))
        val disc1Id = dao.inserirDisciplina(Disciplina("Programação Mobile"))
        val disc2Id = dao.inserirDisciplina(Disciplina("Inteligência Artificial"))
        // 2. Inserir Estudante
        val estId = dao.inserirEstudante(Estudante("Ana", cursoId))
        // 3. Associar estudante às disciplinas (Tabela de Junção N:N)
        dao.inserirEstudanteDisciplina(EstudanteDisciplinaCrossRef(estId, disc1Id))
        dao.inserirEstudanteDisciplina(EstudanteDisciplinaCrossRef(estId, disc2Id))
        // 4. Buscar e registrar Estudante com suas Disciplinas
        val ec = dao.obterEstudanteComDisciplinas(estId)
        ec?.let {
            Log.d("DB_TESTE", "Estudante: ${it.estudante.estudanteNome}")
            for (d in it.disciplinas) {
                Log.d("DB_TESTE", " - Disciplina: ${d.disciplinaNome}")
            }
        }
        // 5. Buscar e registrar Disciplina com seus Estudantes matriculados
        val dc = dao.obterDisciplinaComEstudantes(disc1Id)
        dc?.let {
            Log.d("DB_TESTE", "Disciplina: ${it.disciplina.disciplinaNome}")
            for (e in it.estudantes) {
                Log.d("DB_TESTE", " - Estudante: ${e.estudanteNome}")
            }
        }
    }
}
